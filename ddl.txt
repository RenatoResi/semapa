	id serial4 NOT NULL,
	nome_popular varchar(100) NOT NULL,

	link_foto varchar(200) NULL,

CREATE TABLE public.vistoria (
	id serial4 NOT NULL,
	requerimento_id int4 NOT NULL,

	condicoes text NULL,
	conflitos text NULL,
	risco_queda varchar(10) NULL,
	diagnostico text NULL,
	acao_recomendada varchar(20) NULL,
	tipo_poda text NULL,
	galhos_cortar text NULL,
	medidas_seguranca text NULL,
	observacoes_tecnicas text NULL,
	CONSTRAINT vistoria_pkey PRIMARY KEY (id),
	CONSTRAINT vistoria_especie_id_fkey FOREIGN KEY (especie_id) REFERENCES public.especies(id),
	CONSTRAINT vistoria_requerimento_id_fkey FOREIGN KEY (requerimento_id) REFERENCES public.requerimentos(id) ON DELETE CASCADE,
	CONSTRAINT vistoria_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.users(id) ON DELETE SET NULL
);
